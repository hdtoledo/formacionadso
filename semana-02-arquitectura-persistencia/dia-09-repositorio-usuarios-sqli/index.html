import { STATUS_CODES } from 'node:http';

/**
 * Error de aplicación con código HTTP y código de negocio.
 * Ejemplo (Sesión 09): throw new AppError(409, 'EMAIL_ALREADY_EXISTS', 'El correo ya está registrado.');
 */
export class AppError extends Error {
  constructor(statusCode, code, message, fieldErrors) {
    super(message);
    this.name = 'AppError';
    this.statusCode = statusCode;
    this.code = code;
    this.fieldErrors = fieldErrors;
  }
}

/**
 * Construye SIEMPRE la misma estructura de error para que el frontend la procese igual:
 * { success, statusCode, error, message, code, timestamp, path }
 * El path NO incluye la query string (?email=...), que puede contener datos personales.
 */
export function buildErrorPayload(req, statusCode, code, message, fieldErrors) {
  return {
    success: false,
    statusCode,
    error: STATUS_CODES[statusCode] || 'Error',
    message,
    code,
    ...(fieldErrors ? { fieldErrors } : {}),
    timestamp: new Date().toISOString(),
    path: req.originalUrl.split('?')[0]
  };
}

// 404: ninguna ruta coincidió con la petición
export function notFoundHandler(req, res) {
  res.status(404).json(
    buildErrorPayload(req, 404, 'ROUTE_NOT_FOUND', 'La ruta solicitada no existe en este servidor.')
  );
}

// Manejador global: Express lo reconoce porque recibe 4 parámetros (err, req, res, next)
export function errorHandler(err, req, res, next) {
  if (res.headersSent) return next(err);

  // JSON malformado enviado por el cliente (lo lanza express.json())
  if (err.type === 'entity.parse.failed') {
    return res.status(400).json(
      buildErrorPayload(req, 400, 'INVALID_JSON', 'El cuerpo de la petición no es un JSON válido.')
    );
  }

  // Cuerpo más grande que el límite configurado en express.json({ limit })
  if (err.type === 'entity.too.large') {
    return res.status(413).json(
      buildErrorPayload(req, 413, 'PAYLOAD_TOO_LARGE', 'El cuerpo de la petición supera el tamaño permitido.')
    );
  }

  // Errores controlados lanzados por nuestra propia lógica
  if (err instanceof AppError) {
    return res.status(err.statusCode).json(
      buildErrorPayload(req, err.statusCode, err.code, err.message, err.fieldErrors)
    );
  }

  // Error inesperado: el detalle va al log del servidor, NUNCA al cliente
  console.error('🔥 [Unhandled Exception]:', err);
  return res.status(500).json(
    buildErrorPayload(req, 500, 'INTERNAL_ERROR', 'Ocurrió un error inesperado. Intente nuevamente.')
  );
}

Bloque 1 · Arquitectura

¿Qué es el Patrón Repositorio?

Un repositorio es la única puerta entre la aplicación y una tabla. Los demás archivos le piden datos con métodos con nombre de negocio (findByEmail) y nunca ven SQL.

routes/¿Qué URL y verbo?
controllers/Valida req y arma res
repositories/Único lugar con SQL
config/db.jspool de conexiones
Seguridad auditablePara revisar si hay inyección SQL basta con leer un archivo, no todo el proyecto.
Columnas controladaspassword_hash nunca se selecciona en los métodos públicos: no puede filtrarse por accidente en una respuesta.
Errores traducidosEl código 23505 de PostgreSQL se convierte en un AppError 409 que el frontend entiende.
Bloque 1 · OWASP Top 10: A03 Injection

Anatomía de una Inyección SQL

❌ Código vulnerable (NUNCA escribir así) 
// El valor del usuario se PEGA dentro del texto SQL
const sql = "SELECT email, role FROM users WHERE email = '" + email + "'";
await pool.query(sql);

El atacante envía como email: ' OR '1'='1' --

SELECT email, role FROM users WHERE email = '' OR '1'='1' --'
1. ' cierra la comillaEl texto del atacante deja de ser un dato y pasa a ser código SQL.
2. OR '1'='1' siempre es verdaderoLa condición del WHERE se cumple para todas las filas.
3. -- comenta el restoLa comilla sobrante queda anulada y la consulta es válida.
Resultado medido en PostgreSQL: la consulta concatenada devolvió todas las filas de la tabla users; la parametrizada, 0.
Bloque 1 · Zero Brechas

Defensa: el motor separa el plan de los datos

✅ Consulta parametrizada 
await pool.query(
  'SELECT email, role FROM users WHERE email = $1',
  [email]   // el valor viaja APARTE del SQL
);
Mismo ataque, misma tabla: 0 filas. PostgreSQL buscó un correo cuyo texto literal es ' OR '1'='1' --.
1. Parsepg envía el SQL con el hueco $1. El motor lo analiza y fija su estructura.
2. BindDespués llegan los valores. Ya no pueden cambiar la estructura: solo rellenan el hueco como dato.
3. ExecuteSe ejecuta el plan original. Las comillas del atacante son simples caracteres.
Bloque 1 · Errores críticos en revisiones de código

Trampas que también son inyección

Template literals
`... WHERE email = '${email}'`

Las comillas invertidas también concatenan. Que se vea moderno no lo hace seguro.

Nombres de columna
`ORDER BY ${req.query.sort}`

$1 solo sirve para valores. Para columnas u orden se usa una lista blanca: ['created_at','email'].includes(sort).

"Ya lo validé en el frontend"
curl.exe -X POST -d "{\"email\": 12345}"

Bruno o cURL saltan el formulario. La validación del cliente es comodidad; la seguridad vive en el servidor.

-- ==============================================================================
-- SENA CADPH Garzón · Regional Huila · ADSO
-- Sesión 09 · Hito 0: Migración de endurecimiento del esquema de autenticación
-- Archivo: database/migrations/02_hardening_auth_schema.sql
-- ==============================================================================
-- ¿Por qué una migración y no editar 01_auth_schema.sql?
--   Porque 01 ya se ejecutó en su base de datos (y en la de sus compañeros).
--   Una migración describe el CAMBIO, se ejecuta una sola vez y queda en Git.
--
-- Es IDEMPOTENTE: puede ejecutarse varias veces sin error, y también funciona
-- si su esquema ya tenía la versión corregida de la Sesión 06.
--
-- Ejecución (psql):  \i database/migrations/02_hardening_auth_schema.sql
-- ==============================================================================

BEGIN;

-- -----------------------------------------------------------------------------
-- 1. Tokens: guardar el HASH SHA-256, nunca el token original
-- -----------------------------------------------------------------------------
-- Los tokens existentes estaban en texto plano: se invalidan todos.
DELETE FROM verification_tokens;

DO $$
BEGIN
    IF EXISTS (SELECT 1 FROM information_schema.columns
                WHERE table_name = 'verification_tokens' AND column_name = 'token') THEN
        ALTER TABLE verification_tokens RENAME COLUMN token TO token_hash;
    END IF;

    IF EXISTS (SELECT 1 FROM pg_constraint WHERE conname = 'uq_tokens_token') THEN
        ALTER TABLE verification_tokens RENAME CONSTRAINT uq_tokens_token TO uq_tokens_token_hash;
    END IF;
END $$;

ALTER TABLE verification_tokens ALTER COLUMN token_hash TYPE CHAR(64);

-- -----------------------------------------------------------------------------
-- 2. Índices: eliminar duplicados (UNIQUE ya crea su propio índice B-Tree)
-- -----------------------------------------------------------------------------
DROP INDEX IF EXISTS idx_users_email;
DROP INDEX IF EXISTS idx_tokens_token;
CREATE INDEX IF NOT EXISTS idx_tokens_user_id ON verification_tokens(user_id);

-- -----------------------------------------------------------------------------
-- 3. Correos siempre en minúsculas
-- -----------------------------------------------------------------------------
-- Si dos cuentas solo difieren en mayúsculas, este UPDATE falla por uq_users_email
-- y TODA la migración se deshace (ROLLBACK): resuelva el duplicado y repita.
UPDATE users SET email = LOWER(email) WHERE email <> LOWER(email);

ALTER TABLE users DROP CONSTRAINT IF EXISTS chk_users_email_lower;
ALTER TABLE users ADD CONSTRAINT chk_users_email_lower CHECK (email = LOWER(email));

-- -----------------------------------------------------------------------------
-- 4. updated_at automático en cada UPDATE
-- -----------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION set_updated_at()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = NOW();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_users_updated_at ON users;
CREATE TRIGGER trg_users_updated_at
    BEFORE UPDATE ON users
    FOR EACH ROW
    EXECUTE FUNCTION set_updated_at();

COMMIT;

-- Verificación:  \d verification_tokens   (columna token_hash character(64))
--                \di                      (5 índices)

import { pool } from '../config/db.js';

// Código de PostgreSQL para violación de restricción UNIQUE
const PG_UNIQUE_VIOLATION = '23505';

// El correo se guarda siempre en minúsculas (lo exige chk_users_email_lower de la Sesión 06)
export const normalizeEmail = (email) => email.trim().toLowerCase();

/**
 * Crea un usuario y devuelve sus datos públicos (incluido el UUID generado por PostgreSQL).
 */
async function createUser({ email, passwordHash, role = 'user' }) {
  try {
    const { rows } = await pool.query(
      `INSERT INTO users (email, password_hash, role)
       VALUES ($1, $2, $3)
       RETURNING id, email, role, is_verified, created_at, updated_at`,
      [normalizeEmail(email), passwordHash, role]
    );
    return rows[0];
  } catch (err) {
    if (err.code === PG_UNIQUE_VIOLATION && err.constraint === 'uq_users_email') {
      throw new AppError(409, 'EMAIL_ALREADY_EXISTS', 'El correo ya está registrado.');
    }
    throw err; // Cualquier otro error lo maneja el errorHandler global (500)
  }
}

/**
 * Activa la cuenta y elimina sus tokens de verificación en UNA transacción.
 * Devuelve el usuario activado, o null si no existe o ya estaba verificado.
 *
 * ⚠️ Una transacción necesita UN MISMO cliente: pool.query('BEGIN') no sirve,
 *    porque cada pool.query puede ejecutarse en una conexión distinta del pool.
 */
async function verifyUser(id) {
  const client = await pool.connect();
  try {
    await client.query('BEGIN');

    const { rows } = await client.query(
      `UPDATE users
          SET is_verified = TRUE
        WHERE id = $1 AND is_verified = FALSE
        RETURNING id, email, role, is_verified, created_at, updated_at`,
      [id]
    );

    if (rows.length === 0) {
      await client.query('ROLLBACK');
      return null;
    }

    // Tokens de un solo uso: al activar la cuenta ya no deben servir más
    await client.query('DELETE FROM verification_tokens WHERE user_id = $1', [id]);

    await client.query('COMMIT');
    return rows[0];
  } catch (err) {
    await client.query('ROLLBACK').catch(() => {}); // Si falla algo, se deshace TODO
    throw err;
  } finally {
    client.release(); // Devolver SIEMPRE la conexión al pool (si no, el pool se agota)
  }
}
❌ Error críticoawait pool.query('BEGIN')

Cada pool.query puede usar una conexión distinta: el UPDATE y el DELETE quedarían fuera de la transacción.

pool.connect()Reserva un cliente para que todas las sentencias viajen por la misma conexión.
finally → release()Con max: 10, olvidar release() 10 veces deja la API colgada para siempre.

export async function createDevUser(req, res) {

// PATCH /api/dev/users/:id/verify
export async function verifyDevUser(req, res) {
  const { id } = req.params;
  if (!isUuid(id)) throw invalidId();

  const user = await UserRepository.verifyUser(id);
  if (!user) {
    // verifyUser devuelve null en dos casos: averiguamos cuál para responder bien
    const existing = await UserRepository.findById(id);
    if (!existing) throw userNotFound();
    throw new AppError(409, 'USER_ALREADY_VERIFIED', 'La cuenta ya estaba verificada.');
  }

  res.json({ success: true, data: user });
}
